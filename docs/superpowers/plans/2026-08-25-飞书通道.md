```ts
/**
 * 飞书 SDK 的注入面(2026-08-25,规格 2026-08-25-飞书通道-design.md §四)。
 * real = @larksuiteoapi/node-sdk(锁 1.73.0;WSClient 重连竞态 dsh-im 打过补丁,
 * 真撞到再处理——症状:重连后旧连接回来覆盖新连接,消息重复或丢失);
 * fake = HTTP 到 scripts/fake-feishu-server.mjs(DAWN_FAKE_FEISHU,与 DAWN_FAKE_ILINK 同惯例)。
 */
export interface Feishu凭据 { appId: string; appSecret: string; domain: "feishu" | "lark" }
export interface 设备流回执 { appId: string; appSecret: string; openId: string; domain: "feishu" | "lark" }
export interface 入站消息 { messageId: string; openId: string; text: string; chatType: string; messageType: string }

export interface FeishuSdk {
  /** 设备流建应用:onQr 给二维码 URL;resolve 即 confirmed;reject 带人话(SDK 错误文本不透传——防 secret 进日志) */
  注册应用(声明: { name: string; desc: string }, onQr: (url: string) => void, signal: AbortSignal): Promise<设备流回执>
  /** WS 长连接:onMessage 收 im.message.receive_v1;onState 报 ready/reconnecting/closed。返回停止函数 */
  连接(凭: Feishu凭据, onMessage: (m: 入站消息) => void, onState: (s: "ready" | "reconnecting" | "closed") => void): { stop(): void }
  发文本(凭: Feishu凭据, openId: string, text: string): Promise<void>
  /** 打/撤 Reaction(处理中 OnIt → DONE/ERROR)。失败不抛——它是礼貌不是功能 */
  表情(凭: Feishu凭据, messageId: string, emoji: string, 动作: "create" | "delete"): Promise<void>
}
export function realFeishuSdk(): FeishuSdk   // 动态 import SDK(建会话链路不背它的加载成本)
export function fakeFeishuSdk(baseUrl: string): FeishuSdk
```

real 实现要点(全部照 dsh-feishu 源,解读里有行号):`registerApp({source:"dawn-science", createOnly:true, appPreset:{name,desc}, addons:{preset:false, scopes:{tenant:[SCOPES]}, events:{items:{tenant:["im.message.receive_v1"]}}}})`;SCOPES 常量写死一处 = `["im:message.p2p_msg:readonly","im:message:send_as_bot","im:message.reactions:write_only"]`;`tenant_brand==="lark"` → domain "lark";连接 = `new WSClient({appId,appSecret,domain: domain==="lark"?Lark.Domain.Lark:Lark.Domain.Feishu})` + `EventDispatcher.register({"im.message.receive_v1": …})`,入站取 `message.message_id / sender.sender_id.open_id / JSON.parse(content).text / chat_type / message_type`;发文本 = `client.im.v1.message.create({params:{receive_id_type:"open_id"},data:{receive_id,msg_type:"text",content:JSON.stringify({text})}})`;表情 = `im.v1.messageReaction.create/delete`(delete 需 reaction_id——create 的返回里记下,fake 同形)。
