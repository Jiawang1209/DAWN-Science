# 正式签名与公证 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** mac 包用 Developer ID 签名 + hardened runtime + Apple 公证 + staple，全新机器下载双击就开、以后每次更新 key 都在；
从 ad-hoc 旧版迁过来的那一跳如实告诉用户（界面 + Release 说明 + `startup.log`）；应用内更新换包前验新包签名；Windows 给一条签名的路（默认不走）。

**Architecture:** 基础配置 `electron-builder.yml` 照旧 ad-hoc（本机 pack、`test:packaged`、`rehearse-*` 都坐在它上面），
新增 `electron-builder.release.yml` `extends` 它、写死 Developer ID 身份 + `forceCodeSigning` + `notarize`；
electron-builder 26.15.3 自己签、自己调 `@electron/notarize` 公证并 staple `.app`。判据三层：配置扫描（vitest）→ 构建前环境自检 → 构建后验签脚本（`codesign` / `spctl` / `stapler`，解析是纯函数、有单测）。
运行时读自己的签名身份（解析 `codesign -dv`，与验签脚本同一份解析），与 `userData/signing.json` 比出「ad-hoc → Developer ID」迁移，经 `listCredentials.signing`（协议 8.1）交给界面换措辞。

**Tech Stack:** electron-builder 26.15.3（`app-builder-lib` 的 `MacTargetHelper`）、`@electron/notarize` 2.5.0、Xcode CLT（`codesign` / `spctl` / `xcrun notarytool` / `xcrun stapler`）、GitHub Actions `macos-14`、TypeScript、zod、vitest、Playwright、tsx。

**Spec:** `docs/superpowers/specs/2026-09-27-正式签名与公证-design.md`（D1–D10 是开放决策，未回话按默认；本计划标了每处用的是哪条默认）

**分支：** `signing`，从 `main` 开：`git switch main && git switch -c signing`。

**全局纪律（每个 Task 都适用）**
- **Task 0 没过，后面一个 Task 都不开。** 每个 Task 的第一步「门检」= 照抄 Task 0 Step 2 的四条命令；门没过就停下，告诉作者缺哪样。
- commit 信息写英文，**不加任何 Co-Authored-By 行**；只 commit，**永不 `git push`、永不推 tag、永不 `gh release create`、永不在 GitHub 上点发布**。
  `npm run release:tag` 也不跑——第一个签名版的版本号（D5 默认 0.1.0）与 tag 由作者自己打。
- **秘密一个字都不进仓库、不进日志、不进 commit 信息、不进测试样本**：样本里的 Team ID 一律用 `ABCDE12345`，人名用 `Example Dev`；
  脚本只打印**变量名**，不打印值；本机签名变量只在 `~/.config/dawn-science/signing.env`（仓库外，`chmod 600`）。
- **启动签名版的步骤一律在测试 macOS 用户 `dawnsigntest` 里做**（spec 风险 6：钥匙串项按名字落在登录钥匙串，会碰作者正在用的那一把）。
  `codesign` / `spctl` / `stapler` / `notarytool` 不启动应用，在作者账户里跑。驱动 Electron 时照旧 `DAWN_HIDE_WINDOW=1`（memory：隐藏窗口）。

/**
 * 签名输出的解析与判定（spec `2026-09-27-正式签名与公证-design.md` S4、S6、S7）。
 *
 * **纯函数，不跑命令**：运行时（读自己的签名身份、换包前验新包）与验签脚本（CI）共用这一份——
 * 两处各写一份解析，迟早一处认 `Signature=adhoc`、另一处认 `flags=…adhoc…`，然后判出两个答案。
 *
 * 看不懂的输出答 `unknown`，**不猜**：猜成 developer-id 会让迁移提示说错话，猜成 adhoc 会让换包放过一个没签的包。
 */
export type 签名种类 = "developer-id" | "adhoc" | "unsigned" | "unknown"
export interface 签名详情 {
  种类: 签名种类
  teamId?: string
  identifier?: string
  runtime: boolean
  timestamp: boolean
}

/** 解析 `codesign -dv --verbose=4 <path>` 的 **stderr**（codesign 把详情写在 stderr） */
export function 解析codesign详情(输出: string): 签名详情 {
  const 行 = (k: string) => new RegExp(`^${k}=(.*)$`, "m").exec(输出)?.[1]?.trim()
  const flags = /flags=0x[0-9a-f]+\(([^)]*)\)/.exec(输出)?.[1]?.split(",") ?? []
  const runtime = flags.includes("runtime")
  const timestamp = /^Timestamp=/m.test(输出)
  const identifier = 行("Identifier")
  if (/not signed at all/.test(输出)) return { 种类: "unsigned", runtime: false, timestamp: false }
  if (行("Signature") === "adhoc" || flags.includes("adhoc"))
    return { 种类: "adhoc", ...(identifier ? { identifier } : {}), runtime, timestamp }
  const team = 行("TeamIdentifier")
  if (team && team !== "not set" && /^Authority=Developer ID Application: /m.test(输出))
    return { 种类: "developer-id", teamId: team, ...(identifier ? { identifier } : {}), runtime, timestamp }
  return { 种类: "unknown", runtime, timestamp }
}

/** `spctl -a -vv -t exec`：**只有 `source=Notarized Developer ID` 算公证过**（签了没公证是 `source=Developer ID`） */
export function 判spctl(输出: string): { 通过: boolean; 来源?: string } {
  const 来源 = /^source=(.*)$/m.exec(输出)?.[1]?.trim()
  return { 通过: /: accepted$/m.test(输出) && 来源 === "Notarized Developer ID", ...(来源 ? { 来源 } : {}) }
}

export function 判stapler(输出: string): boolean {
  return /The validate action worked!/.test(输出)
}

/** 发布级的要求，返回**不满足的原因**（空数组 = 过）。每条都是给人看的原话 */
export function 判发布级(d: 签名详情, 期望teamId: string): string[] {
  const 错: string[] = []
  if (d.种类 === "adhoc") 错.push("是 ad-hoc 签名，不是 Developer ID")
  else if (d.种类 !== "developer-id") 错.push(`签名看不懂或没签（${d.种类}）`)
  else if (d.teamId !== 期望teamId) 错.push(`Team ID 是 ${d.teamId}，应为 ${期望teamId}`)
  if (!d.runtime) 错.push("没开 hardened runtime（flags 里没有 runtime）")
  if (d.种类 === "developer-id" && !d.timestamp) 错.push("没有 secure timestamp")
  return 错
}

  return { code: r.status ?? -1, out: `${r.stdout ?? ""}${r.stderr ?? ""}` }
}

    ]

  console.error("release/ 下没有 .app 或 .dmg——先打包")
  process.exit(2)
}

const 失败: string[] = []
const 判 = (名: string, 错: string[]) => {
  if (错.length) 失败.push(`${名}：${错.join("；")}`)

    判("spctl（公证）", s.通过 ? [] : [`source=${s.来源 ?? "?"}`])

    const 文件 = execFileSync("find", [p, "-type", "f", "(", "-perm", "+111", "-o", "-name", "*.node", "-o", "-name", "*.dylib", ")"], { encoding: "utf8" })
      .split("\n").filter(Boolean)
    const 不对的 = 文件.filter((f) => {
      if (!/Mach-O/.test(跑("file", ["-b", f]).out)) return false
      return 解析codesign详情(跑("codesign", ["-dv", "--verbose=4", f]).out).teamId !== team

  } else {
    const v = 跑("codesign", ["--verify", "--verbose=2", p])
    // D8 默认 dmg 不签：这里只记录，不判红；D8 改成签/公证时把下面两行改成 判(...)
    console.log(`  · dmg 签名：${v.code === 0 ? "有" : "无（D8 默认不签）"}`)
    console.log(`  · dmg staple：${判stapler(跑("xcrun", ["stapler", "validate", p]).out) ? "有" : "无（D8 默认不公证）"}`)
  }
}
if (失败.length) {

}

/**
 * 签名环境自检的判定（spec S5）。**只看在不在，只报名字**——这个函数的返回值会进 CI 日志。
 * 公证凭证 electron-builder 认三组（MacTargetHelper.getNotarizeOptions 的顺序：Apple ID → API key → 钥匙串 profile），凑齐一组即可。
 */
type Env = Record<string, string | undefined>
const 有 = (e: Env, k: string) => Boolean(e[k]?.trim())
const 标 = (k: string, ci: boolean) => (ci ? `${k}（GitHub secret ${k === "CSC_LINK" ? "MAC_CSC_LINK" : k === "CSC_KEY_PASSWORD" ? "MAC_CSC_KEY_PASSWORD" : k}）` : k)

export function 缺哪些(e: Env, o: { ci: boolean }): string[] {
  const 缺: string[] = []
  if (o.ci) for (const k of ["CSC_LINK", "CSC_KEY_PASSWORD"]) if (!有(e, k)) 缺.push(标(k, true))
  if (!有(e, "APPLE_TEAM_ID")) 缺.push(o.ci ? "APPLE_TEAM_ID（GitHub variable APPLE_TEAM_ID）" : "APPLE_TEAM_ID")
  const 组 = [
    ["APPLE_ID", "APPLE_APP_SPECIFIC_PASSWORD"],
    ["APPLE_API_KEY", "APPLE_API_KEY_ID", "APPLE_API_ISSUER"],
    ["APPLE_KEYCHAIN_PROFILE"],
  ]
  const 凑齐 = 组.some((g) => g.every((k) => 有(e, k)))
  if (!凑齐) {
    const 半组 = 组.find((g) => g.some((k) => 有(e, k)))
    if (半组) 缺.push(...半组.filter((k) => !有(e, k)).map((k) => 标(k, o.ci)))
    else 缺.push(o.ci ? "公证凭证（APPLE_API_KEY_P8 / APPLE_API_KEY_ID / APPLE_API_ISSUER 三个 secret）" : "公证凭证（APPLE_KEYCHAIN_PROFILE，见 spec P3a）")
  }
  return 缺
}

/**
 * 这个应用自己的签名身份，与「从 ad-hoc 迁到 Developer ID」的判定（spec `2026-09-27-正式签名与公证-design.md` S6）。
 *
 * **为什么要知道**：解不开的 key 在三种情形下该说三种话——迁移那一跳（「最后一次」）、签名版对签名版（「点过拒绝 / 钥匙串被重置」）、
 * 仍是 ad-hoc（「应用更新后钥匙串换了身份」）。说错一种都是在误导人去查不存在的原因。
 *
 * **不在启动路径上**：`codesign` 是一个子进程、几十毫秒，与钥匙串预热同一时机做（2026-08-28 那次几十秒空白的教训）。
 * 读不到就 `unknown`、界面按今天的原句、`startup.log` 记原因——**不猜**。
 *
 * `DAWN_FAKE_SIGNATURE=developer-id:<TEAMID>` / `adhoc`：e2e 与 dev:mock 共用的接缝（准入规则 1），真包里没人设它。
 */
import { existsSync, readFileSync, writeFileSync } from "node:fs"
import { 解析codesign详情, type 签名详情 } from "../update/签名.js"

export type 身份 = 签名详情 & { 原因?: string }
export interface 签名记录 {
  种类: 签名详情["种类"]
  teamId?: string
  /** 迁移那一跳还没收尾（有 key 没重填）。留到 broken 清空为止 */
  迁移中?: true
}

export async function 读签名身份(app路径: string | undefined, 运行: (cmd: string, args: string[]) => Promise<string>): Promise<身份> {
  const 假 = process.env.DAWN_FAKE_SIGNATURE
  if (假) {
    const [种类, teamId] = 假.split(":")
    return 种类 === "developer-id" && teamId
      ? { 种类: "developer-id", teamId, runtime: true, timestamp: true }
      : { 种类: 种类 === "adhoc" ? "adhoc" : "unknown", runtime: true, timestamp: false }
  }
  if (!app路径) return { 种类: "unknown", runtime: false, timestamp: false, 原因: "不是打包版" }
  try {
    return 解析codesign详情(await 运行("/usr/bin/codesign", ["-dv", "--verbose=4", app路径]))
  } catch (e) {
    return { 种类: "unknown", runtime: false, timestamp: false, 原因: e instanceof Error ? e.message : String(e) }
  }
}

export function 判迁移(
  上次: 签名记录 | undefined,
  现在: 身份,
  broken: string[],
): { 记录: 签名记录; transition?: "adhoc-to-developer-id" } {
  const 基 = { 种类: 现在.种类, ...(现在.teamId ? { teamId: 现在.teamId } : {}) }
  if (现在.种类 !== "developer-id" || broken.length === 0) return { 记录: 基 }
  const 从旧来 = !上次 || 上次.种类 === "adhoc" || 上次.迁移中 === true
  return 从旧来 ? { 记录: { ...基, 迁移中: true }, transition: "adhoc-to-developer-id" } : { 记录: 基 }
}

export function 读记录(文件: string): 签名记录 | undefined {
  try {
    return existsSync(文件) ? (JSON.parse(readFileSync(文件, "utf8")) as 签名记录) : undefined
  } catch {
    return undefined // 坏了当没有：最坏是多说一次「最后一次」，不会少说
  }
}
export function 写记录(文件: string, r: 签名记录): void {
  writeFileSync(文件, JSON.stringify(r, null, 2))
}

## 风险（执行时盯着）

1. **Task 1 公证不过**：按 `notarytool log` 定点处理，每个处理写进 `electron-builder.release.yml` 并注明 issue；**不许关 hardened runtime、不许删 `notarize`**。
2. **`test:packaged` 在测试用户里跑不起来**（仓库在作者目录、权限不够）：用 `/Users/Shared` 下的 `git worktree`（`git worktree add /Users/Shared/dawn-sign/repo signing`，在那里 `npm ci`），用完 `git worktree remove`。
3. **M1 的结果可能让 Task 8 的措辞整个作废**（例如签名版能静默读到 ad-hoc 建的钥匙串项，`broken` 根本不会非空）——那就更好：代码不删（它只在 `broken` 非空时说话），Release 说明删掉「重填」那条，历史里照实写。
4. **`codesign -dv` 在用户机器上**：`/usr/bin/codesign` 是系统自带（不需要 Xcode）；拿不到时 `unknown`，界面退回旧措辞——**不许因为读不到身份而把迁移那句说出来**。
5. **两次版本号临时改动**（Task 12 Step 1）一定 `git checkout package.json` 撤掉；`git status` 干净再往下走。
6. **CI 签名路径推之前跑不了**：Task 6 只有静态判据。第一次 tag 失败时的排查顺序写在 `docs/打包与发布.md`：自检那一步列的名字 → `SecKeychainItemImport`（p12/密码）→ 公证日志 → 验签原话。
